Angular Demo

The <barcode-febraban> component binds via @Input() and reacts to input changes via ngOnChanges.

⚠
Angular requires a build step. The interactive demo below uses @allansli/barcode-febraban-core directly (no compilation needed) so you can see the barcode in action right now. For the full Angular integration, see the setup steps and code samples below.
barcode-febraban.component.ts Angular component 
import { Component, Input, OnChanges, SimpleChanges } from "@angular/core";
import { generateBarcodeSequence } from "@allansli/barcode-febraban-core";

@Component({
  selector: "barcode-febraban",
  template: `<div class="barcodei2of5">{{ barcodeSequence }}</div>`
})
export class BarcodeFebrabânComponent implements OnChanges {
  @Input() sequence: string | number = "";

  barcodeSequence: string = "";

  ngOnChanges(changes: SimpleChanges): void {
    if (changes["sequence"]) {
      const seq = String(this.sequence ?? "");
      this.barcodeSequence = seq.length > 0 && !isNaN(Number(seq))
        ? generateBarcodeSequence(seq)
        : "";
    }
  }
}
app.module.ts Module import 
import { NgModule } from "@angular/core";
import { BarcodeFebrabânModule } from "@allansli/ng-barcode-febraban";

@NgModule({
  imports: [BarcodeFebrabânModule]
})
export class AppModule {}
app.component.html Template usage 
<!-- Static -->
<barcode-febraban sequence="1234567890"></barcode-febraban>

<!-- Dynamic — re-renders via ngOnChanges -->
<input [(ngModel)]="sequence" />
<barcode-febraban [sequence]="sequence"></barcode-febraban>

<!-- Extra CSS class and inline styles -->
<barcode-febraban
  [sequence]="sequence"
  className="my-class"
  [style]="{ fontSize: '80px' }"
></barcode-febraban>
Running the Angular app
  1. 1 Install: npm install @allansli/ng-barcode-febraban
  2. 2 Add to angular.json styles:
    "node_modules/@allansli/barcode-febraban-core/assets/css/barcode.css"
  3. 3 Import BarcodeFebrabânModule in your AppModule (or use the component as standalone)
  4. 4 Use <barcode-febraban [sequence]="myValue"></barcode-febraban> in any template
  5. 5 Build the package once: cd packages/angular && npm run build
How it works: The BarcodeFebrabânComponent implements OnChanges to detect input changes. Each time sequence changes, it calls generateBarcodeSequence() from @allansli/barcode-febraban-core and stores the result in barcodeSequence, which Angular's change detection renders in the template.
